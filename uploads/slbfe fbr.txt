import asyncio
from playwright.async_api import async_playwright

async def verify_and_extract_data(passport_no, nic_no):
    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=False)
        context = await browser.new_context()
        page = await context.new_page()

        # 1. Navigation
        print("Navigating to SLBFE FBR Verification...")
        await page.goto("https://applications.slbfe.lk/logroot/agency/cmpl/fbr_sms.asp")

        # 2. Form Submission
        print(f"Submitting details: Passport={passport_no}, NIC={nic_no}")
        await page.fill('input[name="ppno"]', passport_no)
        await page.fill('input[name="idno"]', nic_no)
        
        # Click search and wait for the results page to load
        await asyncio.gather(
            page.wait_for_navigation(),
            page.click('input[name="search"]')
        )

        # 3. Data Extraction from Results Table
        # The results are in a table where the data rows follow the header "Status"
        print("Extracting results...")
        
        results = await page.evaluate('''() => {
            const tables = Array.from(document.querySelectorAll('table'));
            // Find the table that contains "Passport No" text
            const resultTable = tables.find(t => t.innerText.includes('Passport No'));
            if (!resultTable) return [];

            const rows = Array.from(resultTable.querySelectorAll('tr'));
            const extractedData = [];

            // Skip header rows, look for rows with data
            rows.forEach(row => {
                const cells = Array.from(row.cells).map(c => c.innerText.trim());
                // Simple heuristic: if row has 6 cells and the first is a number, it's a data row
                if (cells.length >= 5 && /^\d+$/.test(cells[0])) {
                    extractedData.append({
                        index: cells[0],
                        date_sms: cells[1],
                        passport: cells[2],
                        nic: cells[3],
                        agency_lno: cells[4],
                        status: cells[5]
                    });
                }
            });
            return extractedData;
        }''')

        if results:
            for entry in results:
                print(f"FOUND: {entry['passport']} | Status: {entry['status']} | Date: {entry['date_sms']}")
        else:
            print("No verification records found for the provided details.")

        await browser.close()

if __name__ == "__main__":
    # Example usage
    asyncio.run(verify_and_extract_data('P0262022', '200477304008'))

### Advanced Automation: Batch Processing, Excel Export, and Timeouts

To handle multiple passport checks, save results to Excel, and manage server delays, you need to integrate the `pandas` library and Playwright's built-in timeout settings.

#### Integrated Automation Script

This script processes a list of credentials, handles potential server slowness, and saves everything to a `results.xlsx` file.


`````

import asyncio
import pandas as pd
from playwright.async_api import async_playwright, TimeoutError

async def check_passports(data_list):
    results = []
    
    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=True) # Set to False to watch
        context = await browser.new_context()
        page = await context.new_page()

        # Set a global default timeout (30 seconds)
        page.set_default_timeout(30000)

        for entry in data_list:
            passport = entry['pp']
            nic = entry['nic']
            print(f"Checking: {passport}...")

            try:
                # 1. Navigate to search page
                await page.goto("https://applications.slbfe.lk/logroot/agency/cmpl/fbr_sms.asp", wait_until="networkidle")

                # 2. Fill and Submit
                await page.fill('input[name="ppno"]', passport)
                await page.fill('input[name="idno"]', nic)
                
                # Use a specific timeout for the navigation to handle slow responses
                async with page.expect_navigation(timeout=45000):
                    await page.click('input[name="search"]')

                # 3. Extract Data
                # Locate the result table row containing the passport number
                row_selector = f"tr:has-text('{passport}')"
                
                if await page.locator(row_selector).count() > 0:
                    status_text = await page.locator(row_selector).inner_text()
                    # Split text and grab the 'Recommended' or 'Pending' status (usually last column)
                    status = status_text.split('\t')[-1].strip()
                else:
                    status = "No Record Found"

                results.append({
                    "Passport": passport,
                    "NIC": nic,
                    "Status": status,
                    "Remark": "Success"
                })

            except TimeoutError:
                print(f"Timeout occurred for {passport}. Server might be down.")
                results.append({"Passport": passport, "NIC": nic, "Status": "Error", "Remark": "Server Timeout"})
            except Exception as e:
                print(f"Error checking {passport}: {str(e)}")
                results.append({"Passport": passport, "NIC": nic, "Status": "Error", "Remark": str(e)})

        await browser.close()
    
    # 4. Export to Excel
    df = pd.DataFrame(results)
    df.to_excel("fbr_verification_results.xlsx", index=False)
    print("\nBatch processing complete. Results saved to 'fbr_verification_results.xlsx'.")

# --- Execution ---
if __name__ == "__main__":
    # List of passports/NICs to check
    batch_data = [
        {"pp": "P0262022", "nic": "200477304008"},
        {"pp": "N1234567", "nic": "199012345678"},
        {"pp": "M9876543", "nic": "198587654321"}
    ]
    
    asyncio.run(check_passports(batch_data))
`````


### Root Cause / Logic Explanation

1.  **Looping Logic**: The `for entry in data_list` loop ensures each record is processed sequentially. Using `await page.goto` inside the loop resets the form for every new request.
2.  **Excel Export**: The `pandas.DataFrame` converts the list of dictionaries into a table structure. `to_excel` then handles the file creation using the `openpyxl` engine.
3.  **Timeout Management**: 
    *   `page.set_default_timeout(30000)` sets a 30s limit for all actions.
    *   `page.expect_navigation(timeout=45000)` provides an extra 15s buffer specifically for the database query submission, which is often the slowest part of the SLBFE site.
4.  **Error Handling**: The `try...except TimeoutError` block prevents the entire script from crashing if one specific search fails or the network blips.

import asyncio
import pandas as pd
from playwright.async_api import async_playwright, TimeoutError

        for entry in tqdm(data_list, desc="Verifying Passports", unit="record"):

def job():
    batch_data = [{"pp": "P0262022", "nic": "200477304008"}] # Add your full list here
    asyncio.run(check_batch(batch_data))

# --- Scheduling Logic ---
import schedule
import time

# Schedule for Monday and Friday at 09:00
schedule.every().monday.at("09:00").do(job)
schedule.every().friday.at("09:00").do(job)

print("Scheduler started. Waiting for Monday/Friday 9:00 AM...")
while True:
    schedule.run_pending()
    time.sleep(60) # Check every minute

Running a Python script 24/7 just to wait for a specific time can be resource-heavy. It is better to use **Windows Task Scheduler**:

3.  **Action**: Start a Program.
4.  **Program/script**: `python.exe` (or the full path to your python executable).
5.  **Add arguments**: `C:\path\to\your\script.py`.
